    void func3(string name){
        int vowels=0,consonents=0;
        for(char ch:name){
            ch=tolower(ch);
            if(ch >='a'&& ch<='z'){
                if(ch == 'a' || ch == 'e' || ch == 'i' || ch == 'o' || ch == 'u'){
                    vowels++;
                }
                else{
                    consonents++;
                }
            }
        }
        cout<<"Vowels:"<<vowels<<" Consonents:"<<consonents<<endl;
    }

    /* Problem 4: Count Uppercase, Lowercase, Digits, Spaces, and Special Characters */
    void func4(string name){
        int uppercase=0;
        int lowercase=0;
        int digits=0;
        int spaces=0;
        for(char ch:name){
            if(isupper(ch)) uppercase++;
            else if(islower(ch)) lowercase++;
            else if(isdigit(ch)) digits++;
            else if(isspace(ch)) spaces++;
        }
        cout << "Uppercase:"<<uppercase<<" Lowercase:"<<lowercase<<" Digits:"<<digits<<" Spaces:"<<spaces<<endl;
    }

    void func5(string name){
        /*note : ch in the for loop represents a character as a copy only,

        cout<<"Uppercase string:";
        /* note :thus we used the & operator to bring refernce to the actual values */
        for(char &ch:name){
            if(ch>='a'&& ch<='z'){   
                ch=ch-('a'-'A');
            }

        }

        string result;
        for(int i=str.length()-1;i>=0;i--){
            result.push_back(str[i]);

            cout<<ch<<" ";
        }

            start++;
            end--;
        }
        cout<<endl<<"(Way 2) The string is :"<<endl;
        for(char ch:str){
            cout<<ch<<" ";
        }
    }

    /* Problem 11: Find the Most Frequent Character */
    void func11(){
        string str;
        cout<<"Enter a string :";
        getline(cin,str); 

        //to avoid repetiotion
        vector<bool>visited(str.length(),false);
        int maxCounter=1;

        vector<char>mostFrequentCharacters;
        for(int i=0;i<str.length();i++){
            if(visited[i]) continue;
            int counter=1;
            for(int j=i+1;j<str.length();j++){
                if(str[j]==str[i]){
                    counter++;
                    visited[j]=true;
                }
            }
            if(counter>=maxCounter){
                maxCounter=counter;
                mostFrequentCharacters.push_back(str[i]);
            } 
        }
        cout<<"Most Frequent Character :";
        for(int i=0;i<mostFrequentCharacters.size();i++){
            cout<<mostFrequentCharacters[i]<<" ";
        }
        cout<<endl<<"Frequency :"<<maxCounter<<endl;
    }

        1) way 1 :Create a new string and store the words one by one from behind
        2) way2 : Reverse the complete string as a single word and then re-reverse each 
        inndivisual word
        */

        /* Way 1 */
        string result;
        int start=str.length();
        for(int end=str.length()-1;end>=0;end--){
            if(str[end]==' '||end==0 ){
                for(int i=end+1;i<start;i++){

                }

            };
        }
